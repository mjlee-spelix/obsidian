CREATE OR REPLACE FUNCTION public.spx_log_dify_change()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
AS $fn$
DECLARE
  v_action text;
  v_category text := 'admin';
  v_tenant_id text;
  v_actor_id text;
  v_target_id text;
  v_target_name text;
  v_details jsonb;
BEGIN
  IF TG_OP = 'DELETE' THEN
    v_target_id := OLD.id::text;
  ELSIF TG_OP = 'UPDATE' THEN
    v_target_id := NEW.id::text;
  ELSE
    RETURN NULL;
  END IF;

  CASE TG_TABLE_NAME
    WHEN 'documents' THEN
      IF TG_OP <> 'DELETE' THEN RETURN NULL; END IF;
      v_action := 'document_delete';
      v_target_name := OLD.name;
      v_tenant_id := (SELECT tenant_id::text FROM public.datasets WHERE id = OLD.dataset_id);
      v_details := jsonb_build_object(
        'documentName', OLD.name,
        'datasetId', OLD.dataset_id::text,
        'dataSourceType', OLD.data_source_type,
        'tokens', OLD.tokens
      );

    WHEN 'datasets' THEN
      IF TG_OP <> 'DELETE' THEN RETURN NULL; END IF;
      v_action := 'dataset_delete';
      v_target_name := OLD.name;
      v_tenant_id := OLD.tenant_id::text;
      v_actor_id := OLD.created_by::text;
      v_details := jsonb_build_object(
        'datasetName', OLD.name,
        'description', OLD.description,
        'indexingTechnique', OLD.indexing_technique
      );

    WHEN 'apps' THEN
      IF TG_OP <> 'DELETE' THEN RETURN NULL; END IF;
      v_action := 'app_delete';
      v_target_name := OLD.name;
      v_tenant_id := OLD.tenant_id::text;
      v_actor_id := OLD.created_by::text;
      v_details := jsonb_build_object(
        'appName', OLD.name,
        'mode', OLD.mode,
        'description', OLD.description
      );

    WHEN 'api_tokens' THEN
      IF TG_OP <> 'DELETE' THEN RETURN NULL; END IF;
      v_action := 'api_token_delete';
      v_tenant_id := (SELECT tenant_id::text FROM public.apps WHERE id = OLD.app_id);
      v_target_name := COALESCE(
        (SELECT name FROM public.apps WHERE id = OLD.app_id),
        OLD.id::text
      );
      v_details := jsonb_build_object(
        'tokenType', OLD.type,
        'appId', OLD.app_id::text,
        'lastUsedAt', OLD.last_used_at
      );

    WHEN 'tenant_account_joins' THEN
      IF TG_OP = 'DELETE' THEN
        v_action := 'member_remove';
        v_tenant_id := OLD.tenant_id::text;
        v_target_id := OLD.account_id::text;
        v_actor_id := OLD.account_id::text;
        v_details := jsonb_build_object(
          'role', OLD.role,
          'tenantId', OLD.tenant_id::text
        );
      ELSIF TG_OP = 'UPDATE' AND NEW.role <> OLD.role THEN
        v_action := 'member_role_change';
        v_tenant_id := NEW.tenant_id::text;
        v_target_id := NEW.account_id::text;
        v_actor_id := NEW.account_id::text;
        v_details := jsonb_build_object(
          'oldRole', OLD.role,
          'newRole', NEW.role,
          'tenantId', NEW.tenant_id::text
        );
      ELSE
        RETURN NULL;
      END IF;

    ELSE
      RETURN NULL;
  END CASE;

  BEGIN
    INSERT INTO public.spx_audit_events (
      id, occurred_at, category, action, actor_type, actor_id,
      target_type, target_id, target_name, status, details,
      source, collected_at, tenant_id
    ) VALUES (
      'trg_' || replace(gen_random_uuid()::text, '-', ''),
      NOW(),
      v_category,
      v_action,
      'account',
      v_actor_id,
      TG_TABLE_NAME,
      v_target_id,
      v_target_name,
      'success',
      v_details,
      'pg_trigger',
      NOW(),
      v_tenant_id
    );
  EXCEPTION WHEN OTHERS THEN
    RAISE WARNING 'audit trigger insert failed: %', SQLERRM;
  END;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  ELSE
    RETURN NEW;
  END IF;
END;
$fn$;

KC_ADMIN_PASSWORD=${KEYCLOAK_ADMIN_PASSWORD:-admin}
KC_CLIENT_ID=${AUTH_KEYCLOAK_ID:-dify-audit}

  KC_WAIT=$((KC_WAIT+1))

    echo "[entrypoint] WARNING: Keycloak not reachable after 120s. Skipping client registration."
    KC_WAIT=-1
    break
  fi
  sleep 2
done

  echo "[entrypoint] Fetching Keycloak admin token..."
  TOKEN=$(curl -sf -X POST "${KC_URL}/realms/master/protocol/openid-connect/token" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    -d "username=${KC_ADMIN}&password=${KC_ADMIN_PASSWORD}&grant_type=password&client_id=admin-cli" \
    | jq -r '.access_token // empty')

  if [ -z "$TOKEN" ]; then
    echo "[entrypoint] WARNING: Failed to get Keycloak admin token. Falling back to AUTH_KEYCLOAK_SECRET from env."
  else
    EXISTING_COUNT=$(curl -sf -H "Authorization: Bearer $TOKEN" \
      "${KC_URL}/admin/realms/${KC_REALM}/clients?clientId=${KC_CLIENT_ID}" \
      | jq -r '. | length')

        -H "Authorization: Bearer $TOKEN" \
        -H "Content-Type: application/json" \
        -d "{
          \"clientId\": \"${KC_CLIENT_ID}\",
          \"redirectUris\": [\"${KC_REDIRECT}\"],
          \"webOrigins\": [\"${KC_WEB_ORIGIN}\"],
          \"protocol\": \"openid-connect\",
          \"standardFlowEnabled\": true,
          \"directAccessGrantsEnabled\": false,
          \"serviceAccountsEnabled\": false,
          \"publicClient\": false
        }" >/dev/null
    fi

    CLIENT_UUID=$(curl -sf -H "Authorization: Bearer $TOKEN" \
      "${KC_URL}/admin/realms/${KC_REALM}/clients?clientId=${KC_CLIENT_ID}" | jq -r '.[0].id // empty')
    if [ -n "$CLIENT_UUID" ]; then
      FETCHED_SECRET=$(curl -sf -H "Authorization: Bearer $TOKEN" \
        "${KC_URL}/admin/realms/${KC_REALM}/clients/${CLIENT_UUID}/client-secret" \
        | jq -r '.value // empty')
      if [ -n "$FETCHED_SECRET" ]; then
        export AUTH_KEYCLOAK_SECRET="$FETCHED_SECRET"
        echo "[entrypoint] Keycloak client '${KC_CLIENT_ID}' ready (secret auto-fetched)."
      fi
    fi
  fi
fi
