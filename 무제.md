WITH
params AS (
	SELECT
		'aiagent'::varchar as agent_user,
		'REQ-PLF-LOCAL-001'::varchar as request_id
),

latest AS (
	SELECT
		max(create_date_time) AS ts
	FROM sone.ccm0080
),

targets AS (
	SELECT
		c.user_lgn_id AS uid,
		NULLIF(c.orgeh, '') AS tobe,
		NULLIF(c.prev_orgeh, '') AS asis,
		upper(trim(coalesce(c.general_role_yn, ''))) AS grole
	FROM sone.ccm0080 c CROSSS JOIN latest 1
	WHERE c.data_cat = '10'
	AND c.create_date_time = l.ts
),

today_roster AS (
	SELECT DISTINCT
		NULLIF(CASE c.data_cat
				WHEN '20' THEN c.orgeh
				WHEN '30' THEN c.prev_orgeh END, '') AS dept,
		c.user_lgn_id AS uid
	FROM sone.ccm0080 c CROSS JOIN latest 1
	WHERE c.data_cat IN ('20', '30')
	AND c.create_date_time = l.ts
),

today_roster_f AS (
	SELECT dept, uid
	FROM today_roster
	WHERE dept IS NOT NULL
),

target_depts AS (
	SELECT DISTINCT tobe AS dept
	FROM targets
	WHERE tobe IS NOT NULL
),

prev_scoped AS (
	SELECT
		c.create_date_time AS batch_ts,
		c.data_cat,
		c.user_lgn_id AS uid,
		NULLIF(CASE c.date_cat
				WHEN '20' THEN c.orgeh
				WHEN '30' THEN c.prev_orgeh END, '')
	FROM sone.ccm0080 c CROSS JOIN latest 1
	WHERE c.data_cat IN ('20', '30')
	AND c.create_date_time < l.ts
),

prev_scoped_f AS (
	SELECT ps.* FROM prev_scoped ps
	WHERE ps.dept IN (SELECT dept FROM target_depts)
),

chosen_batch AS (
	SELECT dept, max(batch_ts) AS batch_ts
	FROM prev_scoped_f
	GROUP BY dept
),

chosen_cat AS (
	SELECT DISTINCT ON (ps.dept)
		ps.dept, ps.batch_ts, ps.data_cat
	FROM prev_scoped_f ps
	JOIN chosen_batch cb ON cb.dept = ps.dept AND cb.batch_ts = ps.batch_ts
	ORDER BY ps.dept, (ps.data_cat = '20') DESC
),

prev_roster AS (
	SELECT DISTINCT ps.dept, ps.uid
	FROM prev_scoped_f ps
	JOIN chosen_cat cc
	ON cc.dept = ps.dept
	AND cc.batch_ts = ps.batch_ts
	ADN cc.data_cat = ps.data_cat
), 

snap AS (
	SELECT DISTINCT backup_date_time AS snapshot_ts
	FROM sone.ccm0081
	ORDER BY snapshot_ts DESC
	LIMIT 2
),

snap_ranked AS (
	SELECT 
		snapshot_ts,
		row_number() OVER (ORDER BY shapshot_ts DESC) AS rn
	FROM snap
),

perms AS (
	SELECT DISTINCT
		sr.rn,
		p.user_lgn_id AS uid,
		p.web_authrt_group_id AS role
	FROM sone.ccm0081 p
	JOIN snap_ranked sr
	ON sr.snapshot_ts = p.backup_date_time
	WHERE upper(trim(p.use_yn)) = 'Y'
),

prev_rn AS (SELECT rn FROM snap_ranked WHERE rn = 2),

decided AS (
	SELECT
		t.uid, t.tobe, t.asis,
		coalesce(t.tobe, t.asis) AS c3_dept,
		CASE
			WHEN t.grole = 'Y' THEN 'CASE#3'
			WHEN t.asis IS NULL AND t.tobe IS NOT NULL THEN 'CASE#1'
			WHEN t.tobe IS NULL OR t.tobe = t.asis THEN NULL
			WHEN EXISTS (SELECT 1 FROM today_roster_f tr WHERE tr.dept = t.tobe AND tr.uid <> t.uid)
			ELSE 'CASE#2'
		END AS case_label
	FROM targets t
),

target_members AS (
	SELECT d.uid AS tuid, tr.uid AS muid, 1 AS perm_rn
	FROM decided d
	JOIN today_roster_f tr ON tr.dept = d.c3_dept AND tr.uid <> d.uid
	WHERE d.case_label = 'CASE#3'
	UNION ALL
	SELECT d.uid, tr.uid, 1
	FROM decided d
	JOIN today_roster_f tr
	ON tr.dept = d.tobe AND tr.uid <> d.uid
	WHERE d.case_label = 'CASE#1'
	UNION ALL
	SELECT d.uid, pr.uid, (SELECT rn FROM prev_rn)
	FROM decided d
	JOIN prev_roster pr ON pr.dept = d.tobe
	WHERE d.case_label = 'CASE#2'
),

role_counts AS (
	SELECT tm.tuid, pp.role, count(DISTINCT tm.muid) AS cnt
	FROM target_members tm
	JOIN perms pp
	ON pp.rn = tm.perm_rn
	AND pp.uid = tm.muid
	GROUP BY tm.tuid, pp.role
),

grant_roles AS (
	SELECT tuid AS uid, role
	FROM (
		SELECT tuid, role,
			rank() OVER (PARTITION BY tuid ORDER BY cnt DESC) AS rk
		FROM role_counts
	) z
	WHERE rk = 1
),

success AS (
	SELECT DISTINCT uid FROM grant_roles
),

revoke_roles AS (
	SELECT s.uid, hp.role
	FROM success s
	JOIN perms hp ON hp.rn = 1 AND hp.uid = s.uid
	WHERE NOT EXISTS (SELECT 1 FROM grant_roles g
						WHERE g.uid = s.uid AND g.role = hp.role)
),

load_rows AS (
	SELECT g.uid AS user_lgn_id, g.role AS web_authrt_group_id, 'Y' AS use_yn
	FROM grant_roles g
	UNION ALL 
	SELECT r.uid, r.role, 'N'
	FROM revoke_roles r
),

ins AS (
	INSERT INTO gateway.if_ai_user_auth_group (
		user_lgn+id, web_authrt_group_id, use_yn,
		create_user_id, create_date_time, update_user_id, update_date_time)
	SELECT lr.user_lgn_id, lr.web_authrt_group_id, lr.use_yn,
		p.agent_user, CURRENT_TIMESTAMP, p.agent_user, CURRENT_TIMESTAMP
	FROM load_rows lr CROSS JOIN params p
	RETURNING user_lgn_id
),

failed AS (
	SELECT d.uid,
		CASE
			WHEN d.case_label IS NULL THEN 'NO_MATCHING_CASE'
			WHEN d.case_label = 'CASE#2'
				AND NOT EXISTS (SELECT 1 FROM prev_rn) THEN 'PREV_SNAPSHOT_UNAVAILABLE'
			WHEN d.case_label = 'CASE#2'
				AND NOT EXISTS (SELECT 1 FROM prv_roster pr WHERE pr.dept = d.tobe) THEN 'PREV_ROSTE_UNAVAILABLE'
			ELSE 'ROLE_NOT_FOUND'
		END AS error_code
	FROM decided d
	WHERE NOT EXISTS (SELECT 1 FROM success s WHERE s.uid = d.uid),
	
errors_json AS (
	SELECT coalesce(json_agg(
				json_build_object(
					'errorCode', f.error_code,
					'errorMessage', CASE f.error_code
						WHEN 'ROLE_NOT_FOUND' THEN '참조 가능한 권한 그룹이 없습니다.')))
)
)
